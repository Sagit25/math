\newcommand{\pending}[1]{%
  \par\noindent\fcolorbox{red}{red!4}{%

    \small\textcolor{red}{\textbf{[결과 대기]}}\enspace #1
    \end{minipage}}\par\medskip}

    \end{minipage}}\par\medskip}
